\begin{frame}[fragile]{2. Solve It With if/elif}
\begin{verbatim}
def judge(player, opponent):
    if player == opponent:
        return "tie"
    elif player == "rock" and opponent == "scissors":
        return "win"
    elif player == "scissors" and opponent == "paper":
        return "win"
    elif player == "paper" and opponent == "rock":
        return "win"
    else:
        return "lose"
\end{verbatim}
\begin{itemize}
    \item Five branches. It works, and it matches the map line by line.
    \item Run: \texttt{python3 rps\_if\_elif.py}
\end{itemize}
\note{Live code or run rps\_if\_elif.py. Ask: how many branches? Five. Then ask what we would do for lizard and spock. Plant the worry, do not solve it yet.}
\end{frame}

\begin{frame}[fragile]{3. Solve It With a Dictionary of Dictionaries}
\begin{verbatim}
rules = {
 "rock": {"rock": "tie", "paper": "lose",
          "scissors": "win"},
 "paper": {"rock": "win", "paper": "tie",
           "scissors": "lose"},
 "scissors": {"rock": "lose", "paper": "win",
              "scissors": "tie"},
}
def judge(player, opponent):
    return rules[player][opponent]
\end{verbatim}
\begin{itemize}
    \item \texttt{rules[player][opponent]}: first key picks the row, second picks the column.
    \item Predict: \texttt{rules["paper"]["rock"]}? Run: \texttt{python3 rps\_dict\_of\_dicts.py}
\end{itemize}
\note{Same game, same output. Run both files and diff the demo output: Step 3 scores are identical. The rules are now data you can read as a table, and judge is one line.}
\end{frame}

\begin{frame}{4. Refactoring: Same Game, Better Shape}
\begin{columns}[T]
\column{0.32\textwidth}
\textbf{Got simpler}
\begin{itemize}
    \item \texttt{judge} is one line
    \item No repeated \texttt{and} tests to proofread
    \item The rules read like a table
\end{itemize}
\column{0.34\textwidth}
\textbf{Easier to extend}
\begin{itemize}
    \item New move = new data
    \item Change a rule without touching logic
    \item Loop over the table to check it
\end{itemize}
\column{0.30\textwidth}
\textbf{Stayed the same}
\begin{itemize}
    \item Same inputs and outputs
    \item Same demo results
    \item Same scoreboard idea
\end{itemize}
\end{columns}
\vspace{1ex}
\textbf{Refactoring} = changing the structure without changing the behavior. Verify it: run both versions on all nine matchups.
\note{Ask each column before revealing. Emphasize the test: refactor safely by comparing old and new outputs on every case. That is the Lens 11 Verify habit. Decide/Compare \#1 tie-in: why a dict of dicts and not nine ifs, a list of lists, or one dict keyed by a pair?}
\end{frame}
